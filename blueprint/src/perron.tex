\chapter{Explicit Perron contour tail}

\begin{theorem}{explicit_perron_contour_tail}{0}
For $x,T \ge 2$, $\alpha \in \mathbb{R}$, with $x$ not a positive integer, the difference
between the finite exponential sum over $1 \le n \le x$ and the truncated Perron integral
is bounded by $(x (\log x)^2)/T + \log x$.
\lean{TernaryGoldbach.MinorArcs.explicit_perron_contour_tail}
\uses{perron_horizontal_tail, perron_vertical_tail}
\end{theorem}

\begin{lemma}{perron_horizontal_tail}{0}
The horizontal tail of the truncated Perron integral is bounded by $\log x$.
\lean{TernaryGoldbach.MinorArcs.perron_horizontal_tail}
\end{lemma}

\begin{lemma}{perron_vertical_tail}{0}
The vertical tail of the truncated Perron integral is bounded by $(x (\log x)^2)/T$.
\lean{TernaryGoldbach.MinorArcs.perron_vertical_tail}
\end{lemma}
